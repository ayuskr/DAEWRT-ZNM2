# ============================================================
# ZN-M2 / 京东云亚瑟 AX1800 Pro（IPQ6000，ipq60xx）
#
# 仅编 zn_m2，512M 内存档，不开 WiFi 软件栈。
#
# NSS 显式选中：kmod-nss-wifi-drv 的依赖是
#   @(TARGET_qualcommax_ipq807x || TARGET_qualcommax_ipq60xx) +nss-firmware +kmod-nft-bridge
# 与 ath11k 无关（驱动说明：可在没有 ath11k 的设备上为 raw-IP 提供 NSS 下发）。
# 移除 WiFi 软件栈时它会被连带挤掉，导致状态栏没有 NSS 一行，故在此显式开启。
# PPE 表容量为 ipq60xx 硬件上限 2048，不可调。
# ============================================================

#设备平台
CONFIG_TARGET_qualcommax=y
CONFIG_TARGET_qualcommax_ipq60xx=y
CONFIG_TARGET_MULTI_PROFILE=y
CONFIG_TARGET_PER_DEVICE_ROOTFS=y
CONFIG_TARGET_DEVICE_qualcommax_ipq60xx_DEVICE_zn_m2=y

# 内核配置
CONFIG_KERNEL_ARM_PMU=y

CONFIG_KERNEL_CGROUP_DEVICE=y

CONFIG_KERNEL_NET_CLS_CGROUP=y

# CONFIG_KERNEL_PREEMPT_RCU is not set

# 内存档位（ZN-M2 为 512M）
CONFIG_IPQ_MEM_PROFILE_1024=n
CONFIG_IPQ_MEM_PROFILE_512=y
CONFIG_ATH11K_MEM_PROFILE_1G=n
CONFIG_ATH11K_MEM_PROFILE_512M=y

# ===== NSS（与 WiFi 无关，必须显式开启）=====
CONFIG_PACKAGE_kmod-nss-wifi-drv=y
CONFIG_PACKAGE_nss-firmware=y
CONFIG_PACKAGE_kmod-nft-bridge=y
CONFIG_PACKAGE_kmod-qca-nss-crypto=y
# 状态栏/后台诊断不开启主动探测，避免额外开销
CONFIG_NSS_WIFI_DEBUG=n
